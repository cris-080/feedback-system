import React from 'react';
import { Head, Link } from '@inertiajs/react';

export default function ReportPreview({ reportData, reportId }) {
    const { metadata, demographics, cc_metrics, sqd_metrics, recommendations, harassment, comments } = reportData;

    const exportFormat = metadata.format || 'pdf';
    const isCC = metadata.type === 'cc';

    // SQD Question Mapping
    const sqdTexts = {
        sqd0: "SQD0. I am satisfied with the service that I availed.",
        sqd1: "SQD1. I spent a reasonable amount of time for my transaction.",
        sqd2: "SQD2. The office followed the transaction's requirement from the office or its website.",
        sqd3: "SQD3. The steps (including payment) I needed to do for my transaction were easy and simple.",
        sqd4: "SQD4. I easily found information about my transaction from the office or its website.",
        sqd5: "SQD5. I paid a reasonable amount of fees for my transaction.",
        sqd6: "SQD6. I feel the office was fair to everyone, or \"walang palakasan\" during my transaction.",
        sqd7: "SQD7. I was treated courteously by the staff, and (if I asked for help) the staff was helpful.",
        sqd8: "SQD8. I got what I needed from the government office, or (if denied) denial of request was sufficiently explained to me.",
        overall_satisfaction: "Overall, how would you rate your entire educational experience at CLSU?"
    };

    // Calculate Overall Mean Rating
    const validMeans = Object.values(sqd_metrics).filter(m => parseFloat(m.mean) > 0);
    const overallMean = validMeans.length > 0 
        ? (validMeans.reduce((acc, curr) => acc + parseFloat(curr.mean), 0) / validMeans.length).toFixed(2)
        : '';

    const handlePrint = () => {
        window.print();
    };

    const handleDownloadCSV = () => {
        let csvContent = "data:text/csv;charset=utf-8,";
        
        csvContent += "REPORT METADATA\n";
        csvContent += `Department,${metadata.department}\n`;
        csvContent += `Period,${metadata.period}\n`;
        csvContent += `Total Respondents,${metadata.total_respondents}\n`;
        csvContent += `Standard,${metadata.type.toUpperCase()}\n\n`;

        csvContent += "DEMOGRAPHICS\n";
        csvContent += "Category,Type,Frequency,Percent\n";
        Object.entries(demographics).forEach(([category, data]) => {
            Object.entries(data).forEach(([key, values]) => {
                const safeKey = key.includes(',') ? `"${key}"` : key;
                csvContent += `${category},${safeKey},${values.count},${values.percent}%\n`;
            });
        });
        csvContent += "\n";

        csvContent += "SERVICE QUALITY DIMENSIONS (SQD)\n";
        csvContent += "Code,1,2,3,4,5,N/A,Mean,Adjectival\n";
        Object.entries(sqd_metrics).forEach(([key, data]) => {
            csvContent += `${key},${data.counts['1']},${data.counts['2']},${data.counts['3']},${data.counts['4']},${data.counts['5']},${data.counts['na']},${data.mean},${data.adjectival}\n`;
        });
        csvContent += "\n";

        csvContent += "QUALITATIVE COMMENTS\n";
        comments.forEach(comment => {
            const safeComment = `"${comment.replace(/"/g, '""')}"`;
            csvContent += `${safeComment}\n`;
        });

        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a");
        link.setAttribute("href", encodedUri);
        link.setAttribute("download", `CSM_Data_${metadata.department.replace(/\s+/g, '_')}_${metadata.period.replace(/\s+/g, '_')}.csv`);
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    };

    // --- EXCEL DATA VIEW ---
    if (exportFormat === 'excel') {
        return (
            <div className="min-h-screen bg-gray-100 p-8 font-sans">
                <Head title={`Data Export - ${metadata.department}`} />
                <div className="max-w-7xl mx-auto space-y-6">
                    <div className="flex justify-between items-center bg-white p-5 rounded-lg shadow-sm border border-gray-200">
                        <div>
                            <h2 className="font-bold text-gray-800 text-lg">Raw Data Export Ready</h2>
                            <p className="text-sm text-gray-500">Previewing JSON snapshot for {metadata.department} ({metadata.period})</p>
                        </div>
                        <div className="flex space-x-3">
                            <Link href={route('superadmin.reports.index')} className="bg-white border border-gray-300 text-gray-700 px-4 py-2 rounded shadow-sm hover:bg-gray-50 transition text-sm font-bold">
                                <i className="fa-solid fa-arrow-left mr-2"></i> Back
                            </Link>
                            <button onClick={handleDownloadCSV} className="bg-[#009639] text-white px-5 py-2 rounded shadow-sm hover:bg-[#007a2e] transition text-sm font-bold">
                                <i className="fa-solid fa-download mr-2"></i> Download .CSV File
                            </button>
                        </div>
                    </div>

                    <div className="bg-white p-8 rounded-lg shadow-sm border border-gray-200 overflow-auto max-h-[70vh]">
                        <h3 className="font-bold text-gray-700 mb-4 border-b pb-2">JSON Payload Preview</h3>
                        <pre className="text-xs text-gray-800 bg-gray-50 p-4 rounded border border-gray-100 whitespace-pre-wrap">
                            {JSON.stringify(reportData, null, 4)}
                        </pre>
                    </div>
                </div>
            </div>
        );
    }

    // --- WORD DOCX VIEW ---
    if (exportFormat === 'docx') {
        return (
            <div className="min-h-screen bg-gray-100 p-8 font-sans">
                <Head title={`Word Export - ${metadata.department}`} />
                <div className="max-w-4xl mx-auto space-y-6 mt-10">
                    <div className="flex flex-col items-center justify-center bg-white p-10 rounded-lg shadow-sm border border-gray-200 text-center">
                        <i className="fa-solid fa-file-word text-6xl text-blue-600 mb-4"></i>
                        <h2 className="font-bold text-gray-800 text-2xl mb-2">Official Word Document Ready</h2>
                        <p className="text-gray-500 mb-8 max-w-lg">
                            The system has generated a pixel-perfect .docx file using the official CLSU template for {metadata.department} ({metadata.period}).
                        </p>
                        <div className="flex space-x-4">
                            <Link href={route('superadmin.reports.index')} className="bg-white border border-gray-300 text-gray-700 px-6 py-3 rounded shadow-sm hover:bg-gray-50 transition font-bold">
                                <i className="fa-solid fa-arrow-left mr-2"></i> Back to Archive
                            </Link>
                            <a href={route('superadmin.reports.docx', reportId)} className="bg-blue-600 text-white px-6 py-3 rounded shadow-sm hover:bg-blue-700 transition font-bold">
                                <i className="fa-solid fa-download mr-2"></i> Download Official Word (.docx)
                            </a>
                        </div>
                    </div>
                </div>
            </div>
        );
    }

    // --- STANDARD PDF PRINT VIEW (F57 & F17 MATCH) ---
    return (
        <div className="min-h-screen bg-gray-200 py-10 print:py-0 print:bg-white text-black font-sans leading-tight">
            <Head title={`Report - ${metadata.department}`} />

          {/* Non-Printable Action Bar */}
            <div className="max-w-[210mm] mx-auto mb-4 flex justify-between items-center print:hidden">
                <Link 
                    href={route('superadmin.reports.index')} 
                    className="bg-white border border-gray-300 text-gray-700 px-4 py-2 rounded shadow hover:bg-gray-50 transition font-bold text-sm"
                >
                    <i className="fa-solid fa-arrow-left mr-2"></i> Back to Archive
                </Link>
                <button 
                    onClick={handlePrint}
                    className="bg-[#009639] text-white px-5 py-2 rounded shadow hover:bg-[#007a2e] transition font-bold text-sm flex items-center"
                >
                    <i className="fa-solid fa-print mr-2"></i> Print Document
                </button>
            </div>

            {/* A4 Paper Container */}
            <div className="max-w-[210mm] mx-auto bg-white shadow-2xl print:shadow-none print:max-w-none print:w-full px-12 py-10 print:px-8 print:py-6 box-border text-[11px]">
                
                {/* Official Header */}
                <div className="text-center mb-6 border-b-2 border-green-700 pb-4">
                    <div className="flex items-center justify-center space-x-4 mb-2">
                        {/* Placeholder for CLSU Logo */}
                        <div className="w-16 h-16 rounded-full border-2 border-green-600 flex items-center justify-center text-green-700 font-bold text-xs bg-green-50">CLSU</div>
                        <h1 className="font-serif text-3xl font-bold text-[#006633] tracking-wide">Central Luzon State University</h1>
                    </div>
                    <p className="text-xs">Science City of Muñoz, 3120 Nueva Ecija, Philippines</p>
                    <div className="flex justify-center space-x-4 mt-1 text-xs">
                        <span><i className="fa-solid fa-phone mr-1"></i> (6344) 940-8785</span>
                        <span><i className="fa-solid fa-envelope mr-1"></i> op@clsu.edu.ph</span>
                        <span><i className="fa-solid fa-globe mr-1"></i> clsu.edu.ph</span>
                    </div>
                </div>

                {/* Sub-Header & Title */}
                <div className="text-center mb-4 space-y-1">
                    <p className="uppercase">COLLEGE/OFFICE/DIVISION</p>
                    <p className="font-bold underline uppercase">{metadata.department}</p>
                    {isCC && <p className="font-bold text-sm mt-2 uppercase tracking-widest">CITIZEN'S CHARTER</p>}
                    <h2 className="font-bold text-sm uppercase tracking-widest">OFFICE FEEDBACK REPORT</h2>
                    {/* NEW EVALUATION PERIOD LINE */}
                    <p className="text-xs font-bold mt-1">Evaluation Period: <span className="font-normal underline">{metadata.period}</span></p>
                </div>

                {/* Metadata */}
                <div className="mb-4">
                    <p>This is the summary report on the feedback given by: <span className="font-bold underline text-sm ml-2">{metadata.total_respondents}</span> (student-raters/evaluators)</p>
                </div>

                {/* Demographics Section */}
                <div className="mb-4">
                    <h3 className="font-bold mb-1">Type of Services rendered within the evaluation period</h3>
                    <div className="grid grid-cols-2 gap-6">
                        <table className="w-full border-collapse border border-black text-left">
                            <thead>
                                <tr className="bg-gray-100">
                                    <th className="border border-black p-1">Client Type / Sex</th>
                                    <th className="border border-black p-1 text-center w-20">Frequency</th>
                                    <th className="border border-black p-1 text-center w-20">Percent</th>
                                </tr>
                            </thead>
                            <tbody>
                                <tr><td colSpan="3" className="border border-black p-1 font-bold bg-gray-50 italic">Client Type</td></tr>
                                {Object.entries(demographics.client_type).map(([key, data]) => (
                                    <tr key={`ct-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                                <tr><td colSpan="3" className="border border-black p-1 font-bold bg-gray-50 italic mt-2">Sex</td></tr>
                                {Object.entries(demographics.sex).map(([key, data]) => (
                                    <tr key={`sx-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                            </tbody>
                        </table>

                        <table className="w-full border-collapse border border-black text-left">
                            <thead>
                                <tr className="bg-gray-100">
                                    <th className="border border-black p-1">Region / Service Availed</th>
                                    <th className="border border-black p-1 text-center w-20">Frequency</th>
                                    <th className="border border-black p-1 text-center w-20">Percent</th>
                                </tr>
                            </thead>
                            <tbody>
                                <tr><td colSpan="3" className="border border-black p-1 font-bold bg-gray-50 italic">Region of Residence</td></tr>
                                {Object.entries(demographics.region || {}).map(([key, data]) => (
                                    <tr key={`reg-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                                
                                <tr><td colSpan="3" className="border border-black p-1 font-bold bg-gray-50 italic">Service Availed</td></tr>
                                {Object.entries(demographics.services).map(([key, data]) => (
                                    <tr key={`srv-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                            </tbody>
                        </table>
                    </div>
                </div>

                {/* CC Metrics (Only for F57) */}
                {isCC && (
                    <div className="mb-4">
                        <table className="w-full border-collapse border border-black text-left">
                            <thead>
                                <tr className="bg-gray-100">
                                    <th className="border border-black p-1">Citizen's Charter (CC)</th>
                                    <th className="border border-black p-1 text-center w-20">Frequency</th>
                                    <th className="border border-black p-1 text-center w-20">Percent</th>
                                </tr>
                            </thead>
                            <tbody>
                                <tr><td colSpan="3" className="border border-black p-1 font-bold bg-gray-50">CC1</td></tr>
                                {Object.entries(cc_metrics.cc1).map(([key, data]) => (
                                    <tr key={key}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                                
                                <tr><td colSpan="3" className="border border-black p-1 font-bold bg-gray-50">CC2</td></tr>
                                {Object.entries(cc_metrics.cc2).map(([key, data]) => (
                                    <tr key={`cc2-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}

                                <tr><td colSpan="3" className="border border-black p-1 font-bold bg-gray-50">CC3</td></tr>
                                {Object.entries(cc_metrics.cc3).map(([key, data]) => (
                                    <tr key={`cc3-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                            </tbody>
                        </table>
                    </div>
                )}

                {/* SQD Metrics Matrix */}
                <div className="mb-4 page-break-inside-avoid">
                    <table className="w-full border-collapse border border-black text-[10px] text-center">
                        <thead>
                            <tr className="bg-gray-100">
                                <th rowSpan="2" className="border border-black p-1 text-left w-1/2">Service Quality Dimensions (SQD)</th>
                                <th colSpan="6" className="border border-black p-1">Frequency Ratings</th>
                                <th rowSpan="2" className="border border-black p-1 w-12">MEAN</th>
                                <th rowSpan="2" className="border border-black p-1 w-20">ADJECTIVAL RATING</th>
                            </tr>
                            <tr className="bg-gray-100">
                                <th className="border border-black p-1 w-8">SD (1)</th>
                                <th className="border border-black p-1 w-8">D (2)</th>
                                <th className="border border-black p-1 w-8">N (3)</th>
                                <th className="border border-black p-1 w-8">A (4)</th>
                                <th className="border border-black p-1 w-8">SA (5)</th>
                                <th className="border border-black p-1 w-8">N/A</th>
                            </tr>
                        </thead>
                        <tbody>
                            {Object.entries(sqd_metrics).map(([key, data]) => (
                                <tr key={key}>
                                    <td className="border border-black p-1 text-left">{sqdTexts[key]}</td>
                                    <td className="border border-black p-1">{data.counts['1']}</td>
                                    <td className="border border-black p-1">{data.counts['2']}</td>
                                    <td className="border border-black p-1">{data.counts['3']}</td>
                                    <td className="border border-black p-1">{data.counts['4']}</td>
                                    <td className="border border-black p-1">{data.counts['5']}</td>
                                    <td className="border border-black p-1">{data.counts['na']}</td>
                                    <td className="border border-black p-1 font-bold">{data.mean}</td>
                                    <td className="border border-black p-1">{data.adjectival}</td>
                                </tr>
                            ))}
                            {/* Overall Mean Row */}
                            <tr className="bg-gray-50 font-bold">
                                <td colSpan="7" className="border border-black p-1 text-right">Mean Rating</td>
                                <td className="border border-black p-1">{overallMean}</td>
                                <td className="border border-black p-1"></td>
                            </tr>
                        </tbody>
                    </table>
                </div>

                <div className="grid grid-cols-2 gap-6 mb-4 page-break-inside-avoid">
                    {/* Rating Scale & Harassment */}
                    <div>
                        <div className="mb-4">
                            <p className="font-bold underline mb-1">Rating Scale:</p>
                            <ul className="text-[10px]">
                                <li>4.21 - 5.00 - Excellent</li>
                                <li>3.41 - 4.20 - Very Good</li>
                                <li>2.61 - 3.40 - Good</li>
                                <li>1.81 - 2.60 - Fair</li>
                                <li>1.00 - 1.80 - Needs Improvement</li>
                            </ul>
                        </div>
                        
                        <table className="w-full border-collapse border border-black mb-2">
                            <thead>
                                <tr>
                                    <th className="border border-black p-1 text-left bg-gray-100">Have you experienced any form of harassment during the transactions in this office?</th>
                                    <th className="border border-black p-1 text-center bg-gray-100 w-16">Frequency</th>
                                    <th className="border border-black p-1 text-center bg-gray-100 w-16">Percent</th>
                                </tr>
                            </thead>
                            <tbody>
                                {Object.entries(harassment || {}).map(([key, data]) => (
                                    <tr key={`har-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                            </tbody>
                        </table>
                        <p className="italic text-[10px]">If YES, please specify: _________________________________________</p>
                    </div>

                    {/* Recommendations & Comments */}
                    <div>
                        <table className="w-full border-collapse border border-black mb-4">
                            <thead>
                                <tr>
                                    <th className="border border-black p-1 text-left bg-gray-100">Overall, I would recommend CLSU to my peers</th>
                                    <th className="border border-black p-1 text-center bg-gray-100 w-16">Frequency</th>
                                    <th className="border border-black p-1 text-center bg-gray-100 w-16">Percent</th>
                                </tr>
                            </thead>
                            <tbody>
                                {Object.entries(recommendations).map(([key, data]) => (
                                    <tr key={`rec-${key}`}>
                                        <td className="border border-black p-1 pl-4">{key}</td>
                                        <td className="border border-black p-1 text-center">{data.count}</td>
                                        <td className="border border-black p-1 text-center">{data.percent}%</td>
                                    </tr>
                                ))}
                            </tbody>
                        </table>
                        
                        <div className="h-24">
                            <p className="font-bold underline mb-1">Rater's comments/ Suggestions:</p>
                            <div className="text-[10px] text-gray-800 line-clamp-4">
                                {comments.length > 0 
                                    ? comments.map((comment, index) => <span key={index}>{comment}. </span>) 
                                    : <p className="italic text-gray-500">_________________________________________________________</p>}
                            </div>
                        </div>
                    </div>
                </div>

                {/* Signatories Footer (Mimicking Word Template exactly) */}
                <div className="mt-8 pt-4 page-break-inside-avoid text-xs">
                    <div className="grid grid-cols-2 gap-8 mb-8">
                        <div>
                            <p className="mb-6">Prepared by:</p>
                            <div className="border-b border-black w-3/4 mb-1"></div>
                            <p className="text-[10px] italic">(Unit Head)</p>
                        </div>
                        <div>
                            <p className="mb-6">Attested by:</p>
                            <div className="border-b border-black w-3/4 mb-1"></div>
                            <p className="text-[10px] italic">(VP)</p>
                        </div>
                    </div>
                    <div className="grid grid-cols-2 gap-8">
                        <div>
                            <p className="mb-6">Reviewed by:</p>
                            <div className="border-b border-black w-3/4 mb-1"></div>
                            <p className="text-[10px] italic">(Dean/Director)</p>
                        </div>
                        <div>
                            <p className="mb-6 invisible">Approved by:</p> {/* Spacer to match grid alignment */}
                            <div className="border-b border-black w-3/4 mb-1"></div>
                            <p className="text-[10px] italic">(President)</p>
                        </div>
                    </div>
                </div>

            </div>
        </div>
